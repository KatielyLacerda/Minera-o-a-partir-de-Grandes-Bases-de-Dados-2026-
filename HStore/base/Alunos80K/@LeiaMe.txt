Estes arquivos criam as tabelas de uma base de dados equivalente à base Alunos usada como exemplo nas aulas de Bases de Dados,
mas com uma carga de 80.000 Alunos.

--==    Roda Nomes-Carga.sql                -- Usa Nomes e Sobrenomes.csv --==--  Cria as tabelas de Nomes próprios e Sobrenomes

--==    Roda USPAnuario-Q1.04.sql           --  Usa USPAnuario-Q1.04.csv  --==--  Gera UnidadeUSP
--==    Roda USPAnuario-T3.02.sql           --  Usa USPAnuario-T3.02.csv  --==--  Gera DeptoUSP

--==    Roda Cria-Professor.sql             -->  Só até o bloco indicado: --==--  Cria a tabela Professor80K
--==    Roda Cria-Discip.sql                -->  Só até o bloco indicado: --==--  Cria a tabela Discip80K
--==    Roda Cria-MatrTurma.sql             -->  Só até o bloco indicado: --==--  Cria as tabelas Matricula e Ministra

--==================================================================================================
-- Restrições de Integridade  ======================================================================
--==  As Tabelas estão criadas sem nenhuma restrição de integridade.                            ====
--==  Os índices para criar as restrições de chave primária e segundárias são:                  ====
--==================================================================================================
CREATE UNIQUE INDEX Alunos_PK ON Alunos(NUSP);

CREATE UNIQUE INDEX Prof_PK ON Professor(NNFuncional);

CREATE UNIQUE INDEX Discip_PK ON Discip(Sigla);

CREATE UNIQUE INDEX Turma_PK ON Turma(Codigo);
CREATE UNIQUE INDEX Turma_UK ON Turma(Sigla, Numero);

CREATE INDEX Matric_PK ON Matricula(CodigoTurma, NUSP);
